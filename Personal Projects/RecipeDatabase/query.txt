SELECT root.category_name AS root_name, down1.category_name AS down1_name 
FROM categories AS root 
LEFT OUTER JOIN categories AS down1 
ON down1.parent_category = root.category_id 
WHERE root.parent_category IS null 
ORDER BY root_name, down1_name;
